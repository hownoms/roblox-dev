RUNS = [["localids"], ["localids", "mock"], ["localids", "production"], ["localids", "notstudio"]]


def run_all():
    subprocess.run(["python", "-X", "utf8", "tests/tools/bundle.py"], env=env, capture_output=True, check=True)
    out = {}
    for r in RUNS:
        p = subprocess.run(["luau", "tests/candidate-storage.spec.luau", "-a", *r], env=env,
                           capture_output=True, text=True, encoding="utf-8", errors="replace")
        lines = (p.stdout + p.stderr).splitlines()
        summary = [l for l in lines if l.startswith("CANDIDATE STORAGE") or l.startswith("FAIL")]
        out[" ".join(r)] = (p.returncode, summary)
    return out


killed_all = True
for name, path, old, new in MUTANTS:
    src = open(path, encoding="utf-8", newline="").read()
    assert src.count(old) == 1, (name, old)
    open(path, "w", encoding="utf-8", newline="").write(src.replace(old, new, 1))
    try:
        res = run_all()
    finally:
        open(path, "w", encoding="utf-8", newline="").write(src)
    failing = [k for k, (rc, _) in res.items() if rc != 0]
    print(f"=== MUTANT: {name} ({path})")
    print(f"    {old!r} -> {new!r}")
    for k, (rc, summary) in res.items():
        print(f"    [{k}] exit={rc}")

            print("        " + l)
    print("    RESULT: " + ("KILLED by " + ", ".join(failing) if failing else "SURVIVED"))
